454: 			goto exit;
455: 	} else {
456: 		p = pci_get_device(PCI_VENDOR_ID_VIA, PCI_DEVICE_ID_VIA_8231, NULL);
457: 		if (p == NULL)	/* No problem parts */
458: 			goto exit;
459: 
460: 		/* Check for buggy part revisions */
461: 		if (p->revision < 0x10 || p->revision > 0x12)
462: 			goto exit;
463: 	}
464: 
465: 	/*
466: 	 * Ok we have the problem. Now set the PCI master grant to occur
467: 	 * every master grant. The apparent bug is that under high PCI load
468: 	 * (quite common in Linux of course) you can get data loss when the
469: 	 * CPU is held off the bus for 3 bus master requests.  This happens
470: 	 * to include the IDE controllers....
471: 	 *
472: 	 * VIA only apply this fix when an SB Live! is present but under
